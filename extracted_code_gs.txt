/* ================================================================
   doGet — read endpoints
   ================================================================ */
function doGet(e) {
  // UPDATE: seluruh logic dibungkus try/catch. Sebelumnya, kalau ada error
  // tak terduga di dalamnya (sheet tidak ditemukan, kuota Google habis, dll),
  // request akan gagal TOTAL dan browser menampilkan halaman error generik
  // Google ("Maaf, saat ini tidak dapat membuka file") tanpa pesan error
  // yang jelas — persis gejala yang bikin susah didiagnosa. Sekarang error
  // apa pun akan tetap dikembalikan sebagai JSON yang jelas ke frontend.
  try {
    const action = e.parameter.action || '';

    if (action === 'getVehicle') {
      return jsonResponse({ data: readVehicle() });
    }

    if (action === 'getActiveLoans') {
      return jsonResponse({ data: getActiveLoans() });
    }

    if (action === 'checkVehicleAvailability') {
      return jsonResponse(checkVehicleAvailability());
    }

    if (action === 'getApprovalData') {
      const token = e.parameter.token || '';
      return jsonResponse(getApprovalData(token));
    }

    if (action === 'getSubmissionByNoForm') {
      const noForm = e.parameter.noForm || '';
      return jsonResponse(getSubmissionByNoForm(noForm));
    }

    if (action === 'getSubmissionsList') {
      return jsonResponse({ data: getSubmissionsList() });
    }

    return jsonResponse({ error: 'Unknown action' });
  } catch (err) {
    Logger.log('doGet error: ' + err + '\n' + err.stack);
    return jsonResponse({ error: 'Server error: ' + err.message });
  }
}

/* ================================================================
   doPost — write endpoints
   ================================================================ */
function doPost(e) {
  let body;
  try { body = JSON.parse(e.postData.contents); }
  catch(err) { return jsonResponse({ error: 'Invalid JSON' }); }

  // UPDATE: sama seperti doGet — bungkus semua handler dengan try/catch
  // supaya error tak terduga (mis. MailApp.sendEmail gagal karena kuota
  // email harian habis) tidak membuat seluruh request gagal total tanpa
  // pesan jelas — termasuk kasus data SUDAH tersimpan ke sheet tapi lalu
  // gagal pas mau kirim email approval, yang sebelumnya bikin user
  // mengira SEMUANYA gagal padahal datanya sebenarnya sudah masuk.
  try {
    if (body.action === 'submitForm')        return jsonResponse(handleSubmitForm(body));
    if (body.action === 'submitReturn')      return jsonResponse(handleSubmitReturn(body));
    if (body.action === 'submitApproval')    return jsonResponse(handleSubmitApproval(body));
    if (body.action === 'updateSubmission')  return jsonResponse(handleUpdateSubmission(body));
    if (body.action === 'updateVehicle') {
      if (body.password !== SUPERUSER_PASSWORD) return jsonResponse({ error: 'Unauthorized' });
      writeVehicle(body.data);
      return jsonResponse({ status: 'ok' });
    }

    return jsonResponse({ error: 'Unknown action' });
  } catch (err) {
    Logger.log('doPost error (action=' + body.action + '): ' + err + '\n' + err.stack);
    return jsonResponse({ error: 'Server error: ' + err.message });
  }
}

/* ================================================================
   HELPER: format tanggal sesuai timezone SPREADSHEET (bukan UTC)
   ================================================================
   PENTING: sebelumnya kode ini pakai d.toISOString().slice(0,10) yang
   selalu mengonversi ke UTC. Untuk timezone Indonesia (WIB/WITA/WIT,
   semua di depan UTC), itu bisa membuat tanggal mundur 1 hari.
   ================================================================ */
function fmtDate(v) {
  if (!v) return '';
  const d = new Date(v);
  if (isNaN(d)) return String(v);
  const tz = SpreadsheetApp.getActiveSpreadsheet().getSpreadsheetTimeZone();
  return Utilities.formatDate(d, tz, "yyyy-MM-dd");
}

/* ================================================================
   HELPER: format jam (HH:mm) — bug serupa fmtDate().
   ================================================================
   Input type="time" mengirim string "14:30", tapi Google Sheets
   otomatis mendeteksi & menyimpannya sebagai objek Date/jam internal
   (biasanya bertanggal 1899-12-30 + jam tsb). Kalau dibaca ulang dan
   dikirim mentah ke frontend, hasilnya string ISO panjang yang tidak
   cocok format HH:mm yang dibutuhkan <input type="time">, sehingga
   field itu gagal terisi (kosong) secara diam-diam di browser.
   ================================================================ */
function fmtTime(v) {
  if (!v) return '';
  // Kalau sudah string jam polos "HH:mm" atau "HH:mm:ss", pakai langsung
  if (typeof v === 'string' && /^\d{1,2}:\d{2}/.test(v)) return v.slice(0, 5);
  const d = new Date(v);
  if (isNaN(d)) return String(v);
  const tz = SpreadsheetApp.getActiveSpreadsheet().getSpreadsheetTimeZone();
  return Utilities.formatDate(d, tz, "HH:mm");
}

/* ================================================================
   SUBMIT FORM PEMINJAMAN
   ================================================================ */
function handleSubmitForm(body) {
  const p  = body.pengguna    || {};
  const i  = body.pemakaian   || {};
  const k  = body.kendaraan   || {};
  const s  = body.serahTerima || {};
  const kl = s.kelengkapan    || {};

  // Generate token unik untuk approval link
  const token = Utilities.getUuid();

  // Upload foto SIM ke Google Drive jika ada
  let fotoSimUrl = '';
  let fotoSimFailed = false; // BUG FIX: dulu kalau upload ke Drive gagal,
                              // ini tidak pernah diberitahukan ke frontend —
                              // form tetap tersimpan tapi foto SIM hilang
                              // tanpa peringatan apa pun ke admin.
  if (body.fotoSim && body.fotoSim.startsWith('data:image')) {
    try {
      fotoSimUrl = uploadImageToDrive(body.fotoSim, `SIM_${body.noForm}_${Date.now()}.jpg`);
    } catch(err) {
      Logger.log('Gagal upload foto SIM: ' + err);
      fotoSimFailed = true;
    }
  }

  // Simpan ke sheet Submissions
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_SUBMISSIONS);
  sheet.appendRow([
    new Date(), body.noForm, body.tanggalForm,
    p.nama, p.perusahaan, p.jabatan, p.alamat, p.ktp, p.telepon, p.email,
    i.tujuan, i.tglMulai, i.waktuMulai, i.tglSelesai, i.waktuSelesai,
    i.durasi, i.lokasiJemput, i.lokasiKembali, i.denganDriver,
    k.jenis, k.merk, k.nopol, k.warna, k.tahun, k.vin, k.mesin,
    k.kmAwal, k.levelBensin, k.catatanKondisi,
    s.namaDari, s.jabatanDari, s.tglDari,
    s.namaKe, s.jabatanKe, s.tglKe,
    kl.stnk, kl.kunciCadangan, kl.dongkrak, kl.banSerep, kl.segitiga, kl.p3k, kl.lainnya,
    fotoSimUrl, 'pending', token
  ]);

  // Simpan ke sheet Approvals
  const apSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_APPROVALS);
  apSheet.appendRow([token, body.noForm, 'pending', '', '', '', '', '', '']);

  // Kirim email ke Approver 1
  // UPDATE: dibungkus try/catch sendiri — data form SUDAH tersimpan di
  // baris atas, jadi kalau pengiriman email gagal (kuota MailApp habis,
  // dll), jangan sampai bikin seluruh response error padahal datanya aman.
  let emailFailed = false;
  try {
    sendApprovalEmail(1, token, body, fotoSimUrl);
  } catch(err) {
    Logger.log('Gagal kirim email approval tahap 1: ' + err);
    emailFailed = true;
  }

  // BUG FIX: sebelumnya fotoSimUrl TIDAK ikut dikembalikan ke frontend —
  // frontend cuma tahu "gagal" (fotoSimFailed) tapi tidak pernah punya bukti
  // eksplisit bahwa foto BERHASIL diupload. Sekarang fotoSimUrl selalu
  // disertakan (string kosong kalau memang tidak ada foto dikirim), supaya
  // frontend bisa mengonfirmasi status upload foto secara pasti.
  return { status: 'ok', token: token, fotoSimFailed: fotoSimFailed, fotoSimUrl: fotoSimUrl, emailFailed: emailFailed };
}

/* ================================================================
   UPDATE SUBMISSION (fitur recall/edit data lama dari index.html)
   ================================================================
   Dilindungi PIN (RECALL_PIN). Kolom yang boleh diubah: Tgl Form (3)
   sampai Lainnya-kelengkapan (42). Timestamp, No.Form, Foto SIM URL,
   Approval Status, dan Token TIDAK ikut diubah lewat endpoint ini.

function handleUpdateSubmission(body) {
  if (body.pin !== RECALL_PIN) return { error: 'PIN salah.' };

  const noForm = String(body.noForm || '').trim();
  if (!noForm) return { error: 'No. Form kosong.' };

  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_SUBMISSIONS);
  const data  = sheet.getDataRange().getValues();
  let rowIdx = -1; // 1-indexed row number di sheet
  for (let r = 1; r < data.length; r++) {
    if (String(data[r][1]).trim() === noForm) { rowIdx = r + 1; break; }
  }
  if (rowIdx < 0) return { error: 'No. Form "' + noForm + '" tidak ditemukan.' };

  const p  = body.pengguna    || {};
  const i  = body.pemakaian   || {};
  const k  = body.kendaraan   || {};
  const s  = body.serahTerima || {};
  const kl = s.kelengkapan    || {};

  // Urutan HARUS sama persis dengan kolom 3 (Tgl Form) s.d. 42 (Lainnya), 40 kolom.
  const updatedRow = [
    body.tanggalForm,
    p.nama, p.perusahaan, p.jabatan, p.alamat, p.ktp, p.telepon, p.email,
    i.tujuan, i.tglMulai, i.waktuMulai, i.tglSelesai, i.waktuSelesai,
    i.durasi, i.lokasiJemput, i.lokasiKembali, i.denganDriver,
    k.jenis, k.merk, k.nopol, k.warna, k.tahun, k.vin, k.mesin,
    k.kmAwal, k.levelBensin, k.catatanKondisi,
    s.namaDari, s.jabatanDari, s.tglDari,
    s.namaKe, s.jabatanKe, s.tglKe,
    kl.stnk, kl.kunciCadangan, kl.dongkrak, kl.banSerep, kl.segitiga, kl.p3k, kl.lainnya
  ];

  // Kolom 3 (Tgl Form, 1-indexed) sepanjang 40 kolom sampai kolom 42 (Lainnya)
  sheet.getRange(rowIdx, 3, 1, updatedRow.length).setValues([updatedRow]);

  // Kalau ada foto SIM baru dikirim (fitur re-upload lewat mode edit),
  // upload ke Drive dan timpa kolom Foto SIM URL (kolom 43, 1-indexed).
  let fotoSimFailed = false;
  let fotoSimUrl = '';
  if (body.fotoSim && body.fotoSim.startsWith('data:image')) {
    try {
      fotoSimUrl = uploadImageToDrive(body.fotoSim, `SIM_${noForm}_${Date.now()}.jpg`);
      sheet.getRange(rowIdx, 43).setValue(fotoSimUrl);
    } catch(err) {
      Logger.log('Gagal upload ulang foto SIM: ' + err);
      fotoSimFailed = true;
    }
  }

  // BUG FIX: sama seperti di handleSubmitForm() — sertakan fotoSimUrl di
  // response supaya frontend punya bukti eksplisit upload foto berhasil,
  // bukan cuma asumsi "kalau tidak ada fotoSimFailed berarti sukses".
  return { status: 'ok', noForm: noForm, fotoSimFailed: fotoSimFailed, fotoSimUrl: fotoSimUrl };
}

/* ================================================================
   GET SUBMISSION BY NO.FORM (untuk fitur recall di index.html)
   ================================================================ */
function getSubmissionByNoForm(noForm) {
  noForm = String(noForm || '').trim();
  if (!noForm) return { error: 'No. Form kosong.' };

  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_SUBMISSIONS);
  const data  = sheet.getDataRange().getValues();
  for (let r = 1; r < data.length; r++) {
    if (String(data[r][1]).trim() === noForm) {
      return { data: rowToSubmissionObject(data[r]) };
    }
  }
  return { error: 'No. Form "' + noForm + '" tidak ditemukan.' };
}

/* ================================================================
   GET SUBMISSIONS LIST (untuk daftar pilihan di fitur recall/edit)
   ================================================================
   Mengembalikan ringkasan setiap form (bukan seluruh data) supaya
   ringan, diurutkan dari yang terbaru.
   ================================================================ */
function getSubmissionsList() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_SUBMISSIONS);
  const data  = sheet.getDataRange().getValues();
  const list = [];
  for (let r = 1; r < data.length; r++) {
    const row = data[r];
    const noForm = String(row[1] || '').trim();
    if (!noForm) continue;
    list.push({
      noForm         : noForm,
      nama           : row[3]  || '',
      perusahaan     : row[4]  || '',
      tanggalForm    : fmtDate(row[2]),
      nopol          : row[21] || '',
      approvalStatus : row[43] || ''
    });
  }
  list.reverse(); // terbaru di atas
  return list;
}

/* ================================================================
   SUBMIT APPROVAL (dari halaman approval.html)
   ================================================================ */
function handleSubmitApproval(body) {
  const token    = body.token    || '';
  const role     = parseInt(body.role || 1);
  const decision = body.decision || ''; // 'approve' | 'reject'
  const note     = body.approverNote || '';
  const ttd      = body.ttdApprover  || '';

  if (!token) return { error: 'Token tidak valid' };

  // Update sheet Approvals
  const apSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_APPROVALS);
  const apData  = apSheet.getDataRange().getValues();
  let apRowIdx  = -1;
  for (let r = 1; r < apData.length; r++) {
    if (String(apData[r][0]).trim() === token) { apRowIdx = r + 1; break; }
  }
  if (apRowIdx < 0) return { error: 'Token tidak ditemukan di data approval' };

  const now = new Date().toLocaleString('id-ID');
  let newStatus = '';

  if (role === 1) {
    if (decision === 'approve') {
      apSheet.getRange(apRowIdx, 4).setValue('approved');
      apSheet.getRange(apRowIdx, 5).setValue(now);
      apSheet.getRange(apRowIdx, 6).setValue(note);
      newStatus = 'approved1';
    } else {
      apSheet.getRange(apRowIdx, 4).setValue('rejected');
      apSheet.getRange(apRowIdx, 5).setValue(now);
      apSheet.getRange(apRowIdx, 6).setValue(note);
      newStatus = 'rejected';
    }
  } else if (role === 2) {
    if (decision === 'approve') {
      apSheet.getRange(apRowIdx, 7).setValue('approved');
      apSheet.getRange(apRowIdx, 8).setValue(now);
      apSheet.getRange(apRowIdx, 9).setValue(note);
      newStatus = 'approved2';
    } else {
      apSheet.getRange(apRowIdx, 7).setValue('rejected');
      apSheet.getRange(apRowIdx, 8).setValue(now);
      apSheet.getRange(apRowIdx, 9).setValue(note);
      newStatus = 'rejected';
    }
  }

  // Update status di Approvals row
  apSheet.getRange(apRowIdx, 3).setValue(newStatus);

  // Update status di Submissions
  updateSubmissionStatus(token, newStatus);

  // Ambil data form untuk email
  const formRow = getSubmissionByToken(token);

  // Kirim email sesuai kondisi
  // UPDATE: dibungkus try/catch — status approval SUDAH tersimpan di baris
  // atas, jadi kalau formRow ternyata null (data tidak ketemu) atau
  // pengiriman email gagal, jangan sampai keputusan approve/reject yang
  // sudah tercatat malah dilaporkan "gagal" ke approver.
  let emailFailed = false;
  try {
    if (!formRow) throw new Error('Data submission untuk token ini tidak ditemukan saat kirim email.');
    if (newStatus === 'approved1') {
      // Kirim ke Approver 2
      sendApprovalEmail(2, token, { noForm: formRow[1], pengguna: {
        nama: formRow[3], perusahaan: formRow[4], jabatan: formRow[5], email: formRow[9], telepon: formRow[8]
      }, pemakaian: {
        tujuan: formRow[10], tglMulai: formRow[11], tglSelesai: formRow[13], durasi: formRow[15],
        lokasiJemput: formRow[16], denganDriver: formRow[18]
      }, kendaraan: {
        jenis: formRow[19], merk: formRow[20], nopol: formRow[21], warna: formRow[22], tahun: formRow[23]
      }}, formRow[42]);
    } else if (newStatus === 'approved2') {
      // Kirim email disetujui ke peminjam
      sendResultEmailToBorrower(formRow, 'approved', note);
    } else if (newStatus === 'rejected') {
      // Kirim email ditolak ke peminjam
      sendResultEmailToBorrower(formRow, 'rejected', note);
    }
  } catch(err) {
    Logger.log('Gagal kirim email approval (status=' + newStatus + '): ' + err);
    emailFailed = true;
  }

  return { status: 'ok', newStatus: newStatus, emailFailed: emailFailed };
}

/* ================================================================
   HELPER: cari baris Submissions berdasarkan token
   ================================================================
   Kolom Token = index 44 (0-indexed) / kolom ke-45 (1-indexed).
   ================================================================ */
function getSubmissionByToken(token) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_SUBMISSIONS);
  const data  = sheet.getDataRange().getValues();
  for (let r = 1; r < data.length; r++) {
    if (String(data[r][44]).trim() === token) return data[r];
  }
  return null;
}

function updateSubmissionStatus(token, status) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_SUBMISSIONS);
  const data  = sheet.getDataRange().getValues();
  for (let r = 1; r < data.length; r++) {
    if (String(data[r][44]).trim() === token) {
      sheet.getRange(r + 1, 44).setValue(status); // kolom 44 (1-indexed) = Approval Status
      break;
    }
  }
}

/* ================================================================
   UPLOAD FOTO SIM KE GOOGLE DRIVE
   ================================================================ */
function uploadImageToDrive(base64DataUrl, filename) {
  // Pisahkan header data:image/jpeg;base64, dari data aktual
  const parts    = base64DataUrl.split(',');
  const mimeType = parts[0].match(/:(.*?);/)[1];
  const data     = Utilities.base64Decode(parts[1]);
  const blob     = Utilities.newBlob(data, mimeType, filename);

  // Cari atau buat folder
  let folder;
  const folders = DriveApp.getFoldersByName(DRIVE_FOLDER_NAME);
  if (folders.hasNext()) {
    folder = folders.next();
  } else {
    folder = DriveApp.createFolder(DRIVE_FOLDER_NAME);
  }

  const file = folder.createFile(blob);
  file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  return file.getUrl();
}

/* ================================================================
   ACTIVE LOANS
   ================================================================ */
function getActiveLoans() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const subSheet = ss.getSheetByName(SHEET_SUBMISSIONS);
  const retSheet = ss.getSheetByName(SHEET_RETURNS);

  const retData = retSheet.getDataRange().getValues();
  const returnedForms = new Set();
  for (let r = 1; r < retData.length; r++) {
    const noFormRet = retData[r][3];
    if (noFormRet) returnedForms.add(String(noFormRet).trim());
  }

  const subData = subSheet.getDataRange().getValues();
  if (subData.length < 2) return [];

  const active = [];
  for (let r = 1; r < subData.length; r++) {
    const row = subData[r];
    const noForm = String(row[1] || '').trim();
    if (!noForm || returnedForms.has(noForm)) continue;

    active.push({
      'No.Form'        : noForm,
      'Nama'           : row[3]  || '',
      'Perusahaan'     : row[4]  || '',
      'Jabatan'        : row[5]  || '',
      'Telepon'        : row[8]  || '',
      'Tujuan'         : row[10] || '',
      'Tgl Mulai'      : fmtDate(row[11]),
      'Tgl Selesai'    : fmtDate(row[13]),
      'Durasi'         : row[15] || '',
      'Dengan Driver'  : row[18] || '',
      'Jenis Mobil'    : row[19] || '',
      'Merk/Type'      : row[20] || '',
      'Nopol'          : row[21] || '',
      'Warna'          : row[22] || '',
      'Tahun'          : row[23] || '',
      'KM Awal'        : row[26] || '',
      'Level Bensin'   : row[27] || '',
      'Catatan Kondisi': row[28] || ''
    });
  }

  active.sort((a, b) => (a['Tgl Selesai'] || '9999').localeCompare(b['Tgl Selesai'] || '9999'));
  return active;
}

/* ================================================================
   CHECK VEHICLE AVAILABILITY
   ================================================================ */
function checkVehicleAvailability() {
  const active = getActiveLoans();
  if (active.length === 0) return { isAvailable: true };
  const oldest = active[0];
  return {
    isAvailable  : false,
    borrowerName : oldest['Nama'],
    noForm       : oldest['No.Form'],
    returnDate   : oldest['Tgl Selesai']
  };
}

/* ================================================================

   ================================================================ */
function readVehicle() {
  const sheet  = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_VEHICLE);
  const values = sheet.getDataRange().getValues();
  const headers = values[0];
  const row     = values[1] || [];
  const data    = {};
  headers.forEach((h, i) => data[h] = row[i]);
  return data;
}

function writeVehicle(data) {
  const sheet   = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_VEHICLE);
  const headers = sheet.getDataRange().getValues()[0];
  const row     = headers.map(h => data[h] !== undefined ? data[h] : '');
  sheet.getRange(2, 1, 1, row.length).setValues([row]);
}
